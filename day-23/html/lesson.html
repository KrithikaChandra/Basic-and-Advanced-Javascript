Topic 1 — The Iterator Protocol


const arr = ["a", "b", "c"];

const it = arr[Symbol.iterator]();

console.log(it.next());
console.log(it.next());
console.log(it.next());
console.log(it.next());

for (const ch of arr) {
  console.log(ch);
}

Topic 2 — Make an Object Iterable


const range1 = {
  from: 1,
  to: 5,

  [Symbol.iterator]() {
    let current = this.from;
    const last = this.to;

    return {
      next() {
        if (current <= last) {
          return {
            value: current++,
            done: false
          };
        }

        return {
          value: undefined,
          done: true
        };
      }
    };
  }
};

for (const n of range1) {
  console.log(n);
}

console.log([...range1]);

const [first, ...rest] = range1;

console.log(first, rest);

Topic 3 — Generators with function*


function* simple() {
  yield 1;
  yield 2;
  yield 3;
}

const g = simple();

console.log(g.next());
console.log(g.next());
console.log(g.next());
console.log(g.next());

for (const n of simple()) {
  console.log(n);
}

function* range(from, to) {
  for (let i = from; i <= to; i++) {
    yield i;
  }
}

console.log([...range(1, 5)]);

function* g1() {
  yield "a";
  yield "b";
}

const it1 = g1();

console.log(it1.next().value);
console.log(it1.next().value);
console.log(it1.next().done);

Topic 4 — Infinite Sequences


// Infinite generators are possible:

function* naturals() {
  let n = 1;

  while (true) {
    yield n++;
  }
}

// Use carefully.
// Iterating forever will hang the program.

Topic 5 — yield* Delegation


function* a() {
  yield 1;
  yield 2;
}

function* b() {
  yield 0;
  yield* a();
  yield* [10, 20];
  yield 99;
}

console.log([...b()]);

function* flatten(items) {
  for (const item of items) {
    if (Array.isArray(item)) {
      yield* flatten(item);
    } else {
      yield item;
    }
  }
}

console.log(
  [...flatten([1, [2, [3, [4, 5]]], 6])]
);

Topic 6 — Two-Way Communication with next(value)


function* dialog() {

  const name =
    yield "What's your name?";

  const age =
    yield `Hi ${name}! How old?`;

  return `${name}, ${age}, recorded.`;
}

const g2 = dialog();

console.log(g2.next().value);

console.log(
  g2.next("Priya").value
);

console.log(
  g2.next(25).value
);

Topic 7 — Async Generators & Pagination


function fetchPage(page) {

  const data = {
    1: ["pen", "book", "bag"],
    2: ["mug", "lamp", "fan"],
    3: ["chair", "desk", "rug"],
    4: ["plant", "vase"]
  };

  return Promise.resolve(
    data[page] || []
  );
}

async function* paginate() {

  let page = 1;

  while (true) {

    const items =
      await fetchPage(page);

    if (items.length === 0) {
      return;
    }

    yield* items;

    page++;
  }
}

(async () => {

  for await (
    const item of paginate()
  ) {
    console.log(item);
  }

})();
