Task 3 — Chunk Processing Helper

Build a helper: chunk(array, fn, chunkSize)

Process a large array in smaller chunks and yield control back to the event loop using: setTimeout(..., 0)

Test with a 100k-item array.

View Solution 

function chunk(array, fn, chunkSize) {

    let index = 0;

    function nextChunk() {

        for (
            let i = index;
            i < index + chunkSize && i < array.length;
            i++
        ) {

            fn(array[i]);
        }

        index += chunkSize;

        if (index < array.length) {

            setTimeout(nextChunk, 0);

        } else {

            console.log("Done");
        }
    }

    nextChunk();
}


// Test with 100k items
const arr = Array.from(
    { length: 100000 },
    (_, i) => i
);

chunk(arr, (num) => {

    console.log(num);

}, 1000);

Task 4 — await as a Microtask

Write a snippet proving that: await continues execution as a microtask. Compare:

  • await Promise.resolve()
  • setTimeout(..., 0)

Predict which runs first.

View Solution 

async function testAwait() {

    console.log("Start");

    setTimeout(() => {

        console.log("setTimeout");

    }, 0);

    await Promise.resolve();

    console.log("after await");
}

testAwait();

// after await
