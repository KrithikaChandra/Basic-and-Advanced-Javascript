Task 1 — Tiny EventEmitter

Build createEmitter() returning: { on, off, emit }.

Register two listeners for the event "hello".

Emit the event and verify that both listeners fire.

Remove one listener using off().

Emit again and verify that only one listener fires.

View Solution 

function createEmitter() {
  const events = {};

  return {
    on(event, listener) {
      if (!events[event]) {
        events[event] = [];
      }

      events[event].push(listener);
    },

    off(event, listener) {
      if (!events[event]) return;

      events[event] =
        events[event].filter(
          fn => fn !== listener
        );
    },

    emit(event, ...args) {
      if (!events[event]) return;

      events[event].forEach(
        listener => listener(...args)
      );
    }
  };
}

Task 2 — User Factory

Write a createUser(name, role) factory.

Default role to "user".

The returned object should contain:

  • name
  • role
  • canEdit()

canEdit() should return true only when the role is "admin".

Test with one normal user and one admin user.

View Solution 

function createUser(
  name,
  role = "user"
) {
  return {
    name,
    role,

    canEdit() {
      return this.role === "admin";
    }
  };
}

const user1 =
  createUser("Krithika");

const user2 =
  createUser("John", "admin");

console.log(user1.canEdit());
console.log(user2.canEdit());

Task 3 — Module Cache Factory

Build a createCache() factory using a private Map.

Return: get, set, has, and size.

Create two cache instances and verify that they are independent.

Then export a single cache instance from a module to demonstrate a shared singleton cache.

View Solution 

function createCache() {
  const store = new Map();

  return {
    set(key, value) {
      store.set(key, value);
    },

    get(key) {
      return store.get(key);
    },

    has(key) {
      return store.has(key);
    },

    size() {
      return store.size;
    }
  };
}

const cache1 = createCache();
const cache2 = createCache();

cache1.set(
  "name",
  "Krithika"
);

console.log(
  cache1.get("name")
);

console.log(
  cache2.get("name")
);

Bonus — Reactive Counter with Proxy

Build a reactive(obj, onChange) helper using Proxy.

Whenever a property is updated, call: onChange(prop, value).

Wrap: { count: 0 } and log changes.

Verify that updating both existing and new properties triggers the callback.

View Solution 

function reactive(
  obj,
  onChange
) {
  return new Proxy(obj, {
    set(
      target,
      prop,
      value
    ) {
      target[prop] = value;

      onChange(
        prop,
        value
      );

      return true;
    }
  });
}

const state = reactive(
  { count: 0 },
  (prop, value) => {
    console.log(
      `${prop} → ${value}`
    );
  }
);

state.count = 1;
state.count = 2;
state.username =
  "Krithika";
