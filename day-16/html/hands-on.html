Task 1 — Inspect a Prototype

/* Task 1 Inspect a Prototype

Create const arr = [1, 2, 3];

Use Object.getPrototypeOf(arr) and log it.
Now log:
Object.getPrototypeOf(Object.getPrototypeOf(arr))

Log the next step.
What value ends the chain?

Draw the prototype chain in a comment.
*/
Solution 
const arr = [1, 2, 3];

console.log(Object.getPrototypeOf(arr));

console.log(
  Object.getPrototypeOf(
    Object.getPrototypeOf(arr)
  )
);

console.log(
  Object.getPrototypeOf(Object.prototype)
);

// arr
// ↓
// Array.prototype
// ↓
// Object.prototype
// ↓
// null

Task 2 — Build with Object.create()

/* Task 2 Build with Object.create()

Create a vehicle object with:

start() {
  console.log(`${this.name} starting`);
}

Create:
car.name = "Tata Nexon"
bike.name = "Royal Enfield"

Call start() on both.

Use:
hasOwnProperty()
and in

to check:
- own properties
- inherited properties
*/
Solution 
const vehicle = {
  start: function () {
    console.log(`${this.name} starting`);
  }
};

const car = Object.create(vehicle);
car.name = "Tata Nexon";

const bike = Object.create(vehicle);
bike.name = "Royal Enfield";

car.start();
bike.start();

console.log(car.hasOwnProperty("name"));
console.log(car.hasOwnProperty("start"));

console.log("start" in car);

Task 3 — Constructor Function Inheritance

/* Task 3 Constructor Function Inheritance

Create Person(name)
and set:
this.name = name

Add:
Person.prototype.greet()

Create Student(name, school)

Call:
Person.call(this, name)

Inherit using:
Object.create(Person.prototype)

Add:
study()

Create:
new Student("Riya", "IIT Delhi")

Call:
greet() and study()
*/
Solution 
function Person(name) {
  this.name = name;
}

Person.prototype.greet = function () {
  console.log("Hi, I'm " + this.name);
};

function Student(name, school) {
  Person.call(this, name);
  this.school = school;
}

Student.prototype =
  Object.create(Person.prototype);

Student.prototype.constructor = Student;

Student.prototype.study = function () {
  console.log(
    this.name +
    " studies at " +
    this.school
  );
};

const student1 =
  new Student("Riya", "IIT Delhi");

student1.greet();
student1.study();

Bonus — hasOwnProperty vs in

/* Bonus

Create:

const dog = Object.create({
  species: "Canis"
});

dog.name = "Bruno";

Predict results for:

dog.hasOwnProperty("name")
dog.hasOwnProperty("species")

"name" in dog
"species" in dog
"toString" in dog

Write the rule for:
hasOwnProperty vs in
*/
Solution 
const dog = Object.create({
  species: "Canis"
});

dog.name = "Bruno";

console.log(
  dog.hasOwnProperty("name")
); // true

console.log(
  dog.hasOwnProperty("species")
); // false

console.log("name" in dog); // true
console.log("species" in dog); // true
console.log("toString" in dog); // true

// hasOwnProperty()
// → own properties only

// in
// → own + inherited
