Task 1 — Inspect a Prototype

Create: const arr = [1, 2, 3]

Use: Object.getPrototypeOf() to inspect the prototype chain.

Find what value ends the chain.

View Solution 

const arr = [1, 2, 3];

console.log(Object.getPrototypeOf(arr));

console.log(Object.getPrototypeOf(Object.getPrototypeOf(arr)));

console.log(Object.getPrototypeOf(Object.prototype));

// Prototype chain:
// arr
// ↓
// Array.prototype
// ↓
// Object.prototype
// ↓
// null

Task 2 — Build with Object.create()

Create a vehicle object with a start() method.

Use Object.create() to create:

  • car → Tata Nexon
  • bike → Royal Enfield

Verify: name is own property and start is inherited.

View Solution 

const vehicle = {
  start: function () {

    console.log(`${this.name} starting`);
  }
};

const car = Object.create(vehicle);
car.name = "Tata Nexon";

const bike = Object.create(vehicle);
bike.name = "Royal Enfield";

car.start();
bike.start();

// Own property checks
console.log(car.hasOwnProperty("name"));

console.log(car.hasOwnProperty("start"));

// Inherited property checks
console.log("start" in car);

Task 3 — Constructor Function Inheritance

Build:

  • Person(name)
  • Student(name, school)

Use:

  • Person.call()
  • Object.create()
  • prototype inheritance
View Solution 

function Person(name) {
this.name = name;
}

Person.prototype.greet = function () {

  console.log("Hi, I'm " + this.name);
};

function Student(name, school) {

  Person.call(this, name);
  this.school = school;
}

Student.prototype =Object.create(Person.prototype);

Student.prototype.constructor =Student;

Student.prototype.study =
  function () {
    console.log(this.name +" studies at " +this.school);
  };

const student1 =new Student("Riya","IIT Delhi");

student1.greet();
student1.study();

Bonus — hasOwnProperty vs in

Create:

const dog = Object.create({ species: "Canis" })

Add: dog.name = "Bruno"

Test:

  • hasOwnProperty()
  • in operator
View Solution 

const dog = Object.create({species: "Canis"});

dog.name = "Bruno";


console.log(dog.hasOwnProperty("name"));// true

console.log(dog.hasOwnProperty("species")); // false

console.log("name" in dog);// true

console.log("species" in dog);// true

console.log("toString" in dog);// true

// Rule:
// hasOwnProperty()
// checks ONLY own properties.
//
// in
// checks own + inherited properties.
