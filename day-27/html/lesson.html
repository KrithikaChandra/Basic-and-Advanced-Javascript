Topic 1 — Debounce vs Throttle


// DEBOUNCE — only fire after delay ms of silence
function debounce(fn, delay) {
  let timer;

  return function (...args) {
    clearTimeout(timer);
    timer = setTimeout(
      () => fn.apply(this, args),
      delay
    );
  };
}

// THROTTLE — fire at most once per delay ms
function throttle(fn, delay) {
  let lastCall = 0;

  return function (...args) {
    const now = Date.now();

    if (now - lastCall < delay) return;

    lastCall = now;
    fn.apply(this, args);
  };
}


// Debounce:
// Search box, autocomplete, resize events

// Throttle:
// Scroll, mousemove, resize tracking

Topic 2 — Memory Leaks


// Common causes of memory leaks:

// 1. Forgotten globals
function leaky() {
  user = { name: "Priya" };
}

// 2. Forgotten intervals
const id = setInterval(() => {
  doStuff();
}, 1000);

clearInterval(id);

// 3. Detached DOM listeners
const btn = document.querySelector("#save");

btn.addEventListener("click", handler);

btn.remove();

// 4. Closures holding large data
function makeHandler(largeData) {
  return () => largeData.length;
}


// Fixes:
// - Use let/const
// - Clear timers
// - Remove listeners
// - Avoid unnecessary references
