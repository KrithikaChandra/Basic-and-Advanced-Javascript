Task 1 — Implement Debounce

Write debounce(fn, delay) from scratch.

Use it on a fake search: every "keystroke" should call handleSearch(query).

Simulate:

  • 5 keystrokes 50ms apart
  • 500ms pause
  • 2 more keystrokes

Confirm the search handler only fires twice: once after the first burst and once after the second.

View Solution 

function debounce(fn, delay) {
  let timer;

  return function (...args) {
    clearTimeout(timer);

    timer = setTimeout(() => {
      fn(...args);
    }, delay);
  };
}

function handleSearch(query) {
  console.log("Searching:", query);
}

const debouncedSearch =
  debounce(handleSearch, 200);

Task 2 — Implement Throttle

Write throttle(fn, delay) from scratch.

Simulate a scroll event by calling the throttled function repeatedly.

Verify that the handler runs at most once per delay period.

View Solution 

function throttle(fn, delay) {
  let lastCall = 0;

  return function (...args) {
    const now = Date.now();

    if (now - lastCall >= delay) {
      lastCall = now;
      fn(...args);
    }
  };
}

function onScroll() {
  console.log("Scroll handled");
}

const throttledScroll =
  throttle(onScroll, 200);

Task 3 — Spot the Memory Leaks

Read each snippet and label it: Leak or Safe.

  1. setInterval(updateClock, 1000) inside a component that unmounts.
  2. Add an event listener, then remove the DOM node.
  3. msg = "hello" without let or const.
  4. React cleanup with clearInterval().

Explain why each is safe or unsafe.

View Solution 

// (a) Leak
setInterval(updateClock, 1000);

// (b) Leak
btn.addEventListener("click", handler);
btn.remove();

// (c) Leak
function () {
  msg = "hello";
}

// (d) Safe
useEffect(() => {
  const id = setInterval(...);

  return () =>
    clearInterval(id);
}, []);

Bonus — Sanitise User Input

Write a function: safeText(html).

It should strip all HTML tags and return plain text only.

Test using:

<img src=x onerror="alert(1)"> hello

Insert the result using textContent and confirm that no script executes.

View Solution 

function safeText(html) {
  const temp =
    document.createElement("div");

  temp.innerHTML = html;

  return temp.textContent || "";
}

const result = safeText(
  '<img src=x onerror="alert(1)"> hello'
);

console.log(result);
