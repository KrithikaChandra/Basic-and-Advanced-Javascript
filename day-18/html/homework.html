Task 1 — wait(ms) Promise

Write a wait(ms) function that returns a Promise resolving after ms milliseconds.

Use: 


wait(500)
.then(() => wait(500))
.then(() => console.log("1s"));

Verify that "1s" logs after approximately 1 second.

View Solution 

function wait(ms) {

    return new Promise((resolve) => {

        setTimeout(() => {
            resolve();
        }, ms);

    });
}

wait(500)

.then(() => wait(500))

.then(() => console.log("1s"));

Task 2 — Sequential Promise Chaining

Create a fetchData(id) function that randomly resolves or rejects (50/50 chance).

Chain three sequential .then() calls and end with .catch().

Run multiple times to observe both success and failure flows.

View Solution 

function fetchData(id) {

    return new Promise((resolve, reject) => {

        setTimeout(() => {

            const success = Math.random() > 0.5;

            if (success) {

                resolve(`Data fetched for ID: ${id}`);

            } else {

                reject(`Failed to fetch data for ID: ${id}`);
            }

        }, 1000);

    });
}

fetchData(1)

.then((result1) => {

    console.log(result1);
    return fetchData(2);

})

.then((result2) => {

    console.log(result2);
    return fetchData(3);

})

.then((result3) => {

    console.log(result3);
    console.log("All requests completed successfully");

})

.catch((error) => {

    console.log("Error occurred:", error);

});

Task 3 — Promise.race Timeout

Use Promise.race() to add a timeout to a fetch request.

Race:

  • fetchUser(7)
  • wait(2000).then(() => Promise.reject(new Error("timeout")))

Verify that a timeout error occurs if the fetch takes too long.

View Solution 

function wait(ms) {

    return new Promise((resolve) => {

        setTimeout(resolve, ms);

    });
}

function fetchUser(id) {

    return new Promise((resolve) => {

        setTimeout(() => {

            resolve(`User ${id} fetched successfully`);

        }, 3000);

    });
}

Promise.race([

    fetchUser(7),

    wait(2000).then(() =>
        Promise.reject(new Error("timeout"))
    )

])

.then((result) => {

    console.log(result);

})

.catch((error) => {

    console.error(error.message);

});

Task 4 — Promise.any()

Use Promise.any() with three flaky fetch requests.

Demonstrate that it returns the first successful Promise while ignoring rejected ones.

If all Promises fail, handle the error.

View Solution 

function flakyFetch(endpoint) {

    return new Promise((resolve, reject) => {

        const success = Math.random() > 0.5;

        if (success) {

            resolve(`Data received from ${endpoint}`);

        } else {

            reject(`Network error at ${endpoint}`);
        }

    });
}

const p1 = flakyFetch("/user");
const p2 = flakyFetch("/orders");
const p3 = flakyFetch("/items");

Promise.any([p1, p2, p3])

.then((firstSuccess) => {

    console.log("Got one:", firstSuccess);

})

.catch(() => {

    console.error("Everyone failed");

});
