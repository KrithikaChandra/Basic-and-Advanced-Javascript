Task 1 — Sync vs Async Output

Predict the output order before running:

  • console.log("A")
  • setTimeout(..., 0)
  • Promise.resolve().then(...)

Run the code and compare the actual output.

Explain why Promise callbacks run before setTimeout(..., 0).

View Solution 

console.log("A");

setTimeout(() => 
    console.log("B"), 0
);

console.log("C");

Promise.resolve().then(() => 
    console.log("D")
);

// Output:
// A
// C
// D
// B

// D runs before B because:
// Promise callbacks go into the microtask queue.
// setTimeout callbacks go into the macrotask queue.
// Microtasks always run first.

Task 2 — Promisify a Callback API

Convert a callback-style function into a Promise-based function.

Wrap:

delayLog(msg, ms, cb)

into:

delayLogPromise(msg, ms)

Chain three delayed logs:

  • "1" after 300ms
  • "2" after 200ms
  • "3" after 100ms
View Solution 

function delayLog(msg, ms, cb) {

    setTimeout(() => {

        console.log(msg);
        cb(null);

    }, ms);
}

// Promisified version
function delayLogPromise(msg, ms) {

    return new Promise((resolve, reject) => {

        delayLog(msg, ms, (err) => {

            if (err) {
                reject(err);
            } else {
                resolve();
            }

        });

    });
}

delayLogPromise("1", 300)

    .then(() => delayLogPromise("2", 200))

    .then(() => delayLogPromise("3", 100))

    .catch(err => console.log(err));

Class Work — Fetch Single Post with Promises

Use fetch() with Promise chaining.

Fetch a single post from:

jsonplaceholder.typicode.com

Log:

  • Post title
  • Post body

Handle errors using .catch().

View Solution 

function fetchPostWithPromise(postId) {

    fetch(
        `https://jsonplaceholder.typicode.com/posts/${postId}`
    )

    .then(response => {

        if (!response.ok) {

            throw new Error(
                `HTTP error! status: ${response.status}`
            );
        }

        return response.json();
    })

    .then(post => {

        console.log("Post:", post.title);
        console.log("Content:", post.body);

    })

    .catch(error => {

        console.error(
            "Error fetching post:",
            error.message
        );

    });
}

fetchPostWithPromise(1);

Class Work — Async/Await with Multiple Fetch Calls

Use async/await to:

  • Fetch a user
  • Fetch the user's posts

Log:

  • User name
  • Email
  • Number of posts
  • First post title
View Solution 

async function fetchUserWithPosts(userId) {

    try {

        const userResponse = await fetch(
            `https://jsonplaceholder.typicode.com/users/${userId}`
        );

        if (!userResponse.ok) {
            throw new Error("User not found");
        }

        const user = await userResponse.json();

        const postsResponse = await fetch(
            `https://jsonplaceholder.typicode.com/users/${userId}/posts`
        );

        const posts = await postsResponse.json();

        console.log(
            `User: ${user.name} (${user.email})`
        );

        console.log(
            `Number of posts: ${posts.length}`
        );

        console.log(
            "First post title:",
            posts[0]?.title
        );

        return { user, posts };

    } catch (error) {

        console.error(
            "Error:",
            error.message
        );

    }
}

fetchUserWithPosts(1);

Class Work — Create a New Post

Use fetch() with:

  • POST method
  • JSON request body
  • Async/Await

Log the created post response.

View Solution 

async function createNewPost() {

    try {

        const response = await fetch(
            "https://jsonplaceholder.typicode.com/posts",
            {
                method: "POST",

                headers: {
                    "Content-Type": "application/json",
                },

                body: JSON.stringify({
                    title: "My New Post",
                    body: "This is the content of my post",
                    userId: 1
                })
            }
        );

        const newPost = await response.json();

        console.log(
            "Post created with ID:",
            newPost.id
        );

        console.log(
            "New post:",
            newPost
        );

    } catch (error) {

        console.error(
            "Failed to create post:",
            error
        );

    }
}

createNewPost();

Task 3 — Promise.all in Action

Create a function:

fetchPrice(item, ms)

Use Promise.all to fetch prices in parallel:

  • pen → ₹50
  • book → ₹200
  • bag → ₹800

Measure total execution time using:

Date.now()

Confirm that total time equals the slowest request, not the sum of all delays.

View Solution 

function fetchPrice(item, ms) {

    const prices = {
        pen: 50,
        book: 200,
        bag: 800
    };

    return new Promise((resolve) => {

        setTimeout(() => {

            resolve({
                item,
                price: prices[item]
            });

        }, ms);

    });
}

const t0 = Date.now();

Promise.all([

    fetchPrice("pen", 300),
    fetchPrice("book", 500),
    fetchPrice("bag", 800),

]).then((results) => {

    const total = results.reduce(
        (sum, r) => sum + r.price,
        0
    );

    console.log(`Total: ₹${total}`);

    console.log(
        `Took: ${Date.now() - t0}ms`
    );

});

// Promise.all runs everything concurrently.
// Total time ≈ slowest promise.

Bonus — Promise.all vs Promise.allSettled

Compare:

  • Promise.all()
  • Promise.allSettled()

Use:

  • Two resolved promises
  • One rejected promise

Observe how each behaves.

View Solution 

const p1 = Promise.resolve("ok1");

const p2 = Promise.reject(
    new Error("fail")
);

const p3 = Promise.resolve("ok2");

// Promise.all → fail-fast
Promise.all([p1, p2, p3])

    .then(() => {

        console.log("never reached");

    })

    .catch((err) => {

        console.log(
            "all rejected:",
            err.message
        );

    });

// Promise.allSettled → returns all outcomes
Promise.allSettled([p1, p2, p3])

    .then((results) => {

        results.forEach((r, i) => {

            if (r.status === "fulfilled") {

                console.log(i, "ok:", r.value);

            } else {

                console.log(
                    i,
                    "fail:",
                    r.reason.message
                );
            }

        });

    });

// Use:
// Promise.all → all-or-nothing
// Promise.allSettled → partial failures allowed
