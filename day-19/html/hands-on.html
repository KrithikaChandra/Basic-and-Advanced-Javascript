Task 1 — Convert .then() to async/await

Take this chain: 


fetchUser(7)
.then(u => fetchOrders(u.id))
.then(orders => console.log(orders.length))
.catch(e => console.error(e))

Mock:

  • fetchUser()
  • fetchOrders()

using setTimeout() that resolves in 300ms.

Rewrite the chain as a single async function showOrders(id) using:

  • await
  • try/catch
View Solution 

function fetchUser(id) {

    return new Promise((resolve) => {

        setTimeout(() => {

            resolve({
                id,
                name: "Priya"
            });

        }, 300);
    });
}

function fetchOrders(id) {

    return new Promise((resolve) => {

        setTimeout(() => {

            resolve([
                { id: 1 },
                { id: 2 },
                { id: 3 }
            ]);

        }, 300);
    });
}

async function showOrders(id) {

    try {

        const user = await fetchUser(id);

        const orders =
            await fetchOrders(user.id);

        console.log(orders.length);

    } catch (e) {

        console.error(e);
    }
}

showOrders(7);

Task 2 — Sequential vs Parallel Timing

Create a fetchPrice(id) function that resolves after 500ms with: { id, price: 100 } Write:

  • slow() using separate awaits
  • fast() using Promise.all()

Time both using Date.now().

Confirm:

  • slow() ≈ 1500ms
  • fast() ≈ 500ms
View Solution 

function fetchPrice(id) {

    return new Promise((resolve) => {

        setTimeout(() => {

            resolve({
                id,
                price: 100
            });

        }, 500);
    });
}

async function slow() {

    const t0 = Date.now();

    const a = await fetchPrice(1);
    const b = await fetchPrice(2);
    const c = await fetchPrice(3);

    console.log(
        `Took ${Date.now() - t0}ms`
    );
}

async function fast() {

    const t0 = Date.now();

    const [a, b, c] = await Promise.all([
        fetchPrice(1),
        fetchPrice(2),
        fetchPrice(3),
    ]);

    console.log(
        `Took ${Date.now() - t0}ms`
    );
}

slow();
fast();

Task 3 — Fix the forEach Trap

Take: const ids = [1, 2, 3] and a fetchPrice(id) that resolves after 500ms.

First try: ids.forEach(async (id) => ...) Notice that the function finishes BEFORE the async work completes.

Fix it two ways:

  • for...of
  • Promise.all() + map()
View Solution 

const ids = [1, 2, 3];

function fetchPrice(id) {

    return new Promise((resolve) => {

        setTimeout(() => {

            resolve({
                id,
                price: 100
            });

        }, 500);
    });
}

async function bug() {

    console.log("start bug");

    ids.forEach(async (id) => {

        const p = await fetchPrice(id);

        console.log("got", p);
        console.log("Price", p.price);
    });

    console.log("end bug");
}

async function forOfFix() {

    console.log("start forOf");

    for (const id of ids) {

        const p = await fetchPrice(id);

        console.log("got", p);
        console.log("Price", p.price);
    }

    console.log("end forOf");
}

async function parallel() {

    console.log("start Promise.all");

    const results = await Promise.all(

        ids.map((id) =>
            fetchPrice(id)
        )
    );

    results.forEach((p) => {

        console.log(
            "got Promise.all",
            p.price
        );
    });

    console.log("end");
}

bug();

forOfFix();

parallel();

Bonus — Retry with async/await

Write: retry(fn, attempts) that retries an async function if it fails.

Requirements:

  • Retry up to attempts times
  • Throw the last error if all attempts fail

Test it using a flaky() function that randomly:

  • resolves
  • rejects
View Solution 

async function retry(fn, attempts) {

    let lastError;

    for (let i = 1; i <= attempts; i++) {

        try {

            console.log(`Attempt ${i}`);

            return await fn();

        } catch (error) {

            lastError = error;

            console.log(
                `Failed attempt ${i}: ${error.message}`
            );
        }
    }

    throw lastError;
}

async function flaky() {

    return new Promise((resolve, reject) => {

        setTimeout(() => {

            const success =
                Math.random() > 0.5;

            if (success) {

                resolve("Success!");

            } else {

                reject(
                    new Error("Random failure")
                );
            }

        }, 500);
    });
}

(async () => {

    try {

        const result =
            await retry(flaky, 5);

        console.log(
            "Final Result:",
            result
        );

    } catch (error) {

        console.log(
            "All attempts failed:",
            error.message
        );
    }
})();
