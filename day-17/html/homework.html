Task 1 — Vehicle Inheritance

Build a Vehicle(brand) class with a start() method.

Then build:

  • Car(brand, doors)
  • Bike(brand)

Both should extend Vehicle. Override start() in Car to call super.start() then log "Car-specific check".

Test both classes.

View Solution 

class Vehicle {

    constructor(brand) {
        this.brand = brand;
    }

    start() {
        console.log(`${this.brand} is starting`);
    }
}

class Car extends Vehicle {

    constructor(brand, doors) {
        super(brand);
        this.doors = doors;
    }

    start() {
        super.start();
        console.log("Car-specific check");
    }
}

class Bike extends Vehicle {

    constructor(brand) {
        super(brand);
    }

    start() {
        super.start();
    }
}

const c = new Car("Tata", 4);
c.start();

const b = new Bike("RE");
b.start();

Task 2 — Static Counter

Add a static count property to a Counter class that tracks how many counter objects have been created.

Increment the count inside the constructor.

View Solution 

class Counter {

    static count = 0;

    constructor() {
        Counter.count++;
    }
}

const c1 = new Counter();
const c2 = new Counter();
const c3 = new Counter();

console.log(Counter.count);

Task 3 — Private Temperature Class

Build a Temperature class with a private #celsius field.

Add:

  • Getter for celsius
  • Getter for fahrenheit
  • Setter for celsius

Validate that temperature cannot go below -273.15.

View Solution 

class Temperature {

    #celsius;

    constructor(celsius) {
        this.#celsius = celsius;
    }

    get tempInCelsius() {
        return this.#celsius;
    }

    get tempInFahrenheit() {
        return (this.#celsius * 1.8) + 32;
    }

    set validCelsius(value) {

        if (value < -273.15) {
            throw new Error("Invalid temperature");
        }

        this.#celsius = value;
    }
}

const t = new Temperature(32);

console.log(t.tempInCelsius);
console.log(t.tempInFahrenheit);

t.validCelsius = 40;

console.log(t.tempInCelsius);

Task 4 — Custom ValidationError Class

Convert a basic validation error into a proper class extending Error.

Test that:

  • instanceof ValidationError
  • instanceof Error

both return true.

View Solution 

class ValidationError extends Error {

    constructor(message) {
        super(message);
        this.name = "ValidationError";
    }
}

function validateEmail(email) {

    if (!email.includes("@")) {
        throw new ValidationError(
            "Email must contain @"
        );
    }

    return "Valid Email";
}

try {

    console.log(
        validateEmail("priya-no-at")
    );

} catch (err) {

    console.log(
        err instanceof ValidationError
    );

    console.log(
        err instanceof Error
    );

    if (err instanceof ValidationError) {

        console.log(
            "Validation Error:",
            err.message
        );

    } else {

        console.log(
            "Other Error:",
            err.message
        );
    }
}
