Task 1 — Implement compose()

Implement a compose(...fns) function that executes functions from right to left.

Use the functions: addOne, square, and negate.

Verify: compose(negate, square, addOne)(5) returns -36.

Compare the execution order with pipe().

View Solution 

function compose(...fns) {
  return input =>
    fns.reduceRight((acc, fn) => fn(acc), input);
}

const addOne = x => x + 1;
const square = x => x * x;
const negate = x => -x;

const result = compose(
  negate,
  square,
  addOne
);

console.log(result(5));
// -36

Task 2 — Refactor to a Pure Function

Convert an impure array-sorting function into a pure version.

The original function should mutate the array.

The new version should leave the original array unchanged.

Verify both behaviours in the console.

View Solution 

// Impure
function sortNumbers(arr) {
  return arr.sort((a, b) => a - b);
}

const nums1 = [3, 1, 2];
sortNumbers(nums1);

console.log(nums1);
// [1, 2, 3]

// Pure
function sortNumbersPure(arr) {
  return [...arr].sort((a, b) => a - b);
}

const nums2 = [3, 1, 2];
const sorted = sortNumbersPure(nums2);

console.log(nums2);
// [3, 1, 2]

console.log(sorted);
// [1, 2, 3]

Task 3 — Generic curry()

Implement a generic curry(fn) utility.

It should work with any function that accepts multiple arguments.

Test it with: curry(add)(1)(2)(3).

View Solution 

function curry(fn) {
  return function curried(...args) {

    if (args.length >= fn.length) {
      return fn(...args);
    }

    return (...nextArgs) =>
      curried(...args, ...nextArgs);
  };
}

function add(a, b, c) {
  return a + b + c;
}

const curriedAdd = curry(add);

console.log(
  curriedAdd(1)(2)(3)
);
// 6

Task 4 — Average Order Value Pipeline

Given an array of order objects, calculate the average order value.

Exclude GST and use only: price × quantity.

Build the solution using small pure functions.

Compute the final average value.

View Solution 

const orders = [
  { id: 1, item: "Pen",  price: 50,  quantity: 2 },
  { id: 2, item: "Book", price: 200, quantity: 1 },
  { id: 3, item: "Bag",  price: 800, quantity: 1 },
  { id: 4, item: "Mug",  price: 150, quantity: 3 },
];

const orderValue = order =>
  order.price * order.quantity;

const sum = numbers =>
  numbers.reduce(
    (total, num) => total + num,
    0
  );

const average = numbers =>
  sum(numbers) / numbers.length;

const averageOrderValue =
  average(
    orders.map(orderValue)
  );

console.log(
  averageOrderValue
);
// 387.5
