Topic 1 — Pure Functions


// PURE — no external state, no side effects
function add(a, b) { return a + b; }

function double1(x) {
  return x * 2;
}

function withGST(price, rate = 18) {
  return price * (1 + rate / 100);
}

// IMPURE — depends on external state
let multiplier = 2;

function impureDouble(x) {
  return x * multiplier;
}

// IMPURE — side effect
function logAndDouble(x) {
  console.log(x);
  return x * 2;
}

// IMPURE — mutates argument
function addItem(arr, item) {
  arr.push(item);
  return arr;
}

// PURE version
function addItemPure(arr, item) {
  return [...arr, item];
}

async function processOrder(orderId) {

  const order = await fetchOrder(orderId);

  const validated = validate(order);
  const taxed = applyTax(validated);
  const formatted = formatForDisplay(taxed);

  await saveOrder(formatted);

Topic 3 — Higher-Order Functions


const nums = [1, 2, 3, 4, 5];

nums.map(x => x * 2);
nums.filter(x => x % 2 === 0);
nums.reduce((a, b) => a + b, 0);

function multiplier(factor) {
  return (x) => x * factor;
}

const double = multiplier(2);
const triple = multiplier(3);

console.log(double(5));
console.log(triple(5));

function once(fn) {
  let called = false;
  let result;

  return (...args) => {
    if (called) return result;

    called = true;
    return result = fn(...args);
  };
}

Topic 4 — Compose and Pipe


const compose = (...fns) =>
  (x) => fns.reduceRight((acc, fn) => fn(acc), x);

const pipe = (...fns) =>
  (x) => fns.reduce((acc, fn) => fn(acc), x);

const trim = s => s.trim();
const lower = s => s.toLowerCase();
const split = s => s.split(/\s+/);
const wordCount = arr => arr.length;

const countWords =
  pipe(trim, lower, split, wordCount);

console.log(
  countWords(" Hello World from Jaipur ")
);

Topic 5 — Currying


function add(a, b, c) {
  return a + b + c;
}

function addCurried(a) {
  return function(b) {
    return function(c) {
      return a + b + c;
    };
  };
}

addCurried(1)(2)(3);

const addC =
  (a) => (b) => (c) => a + b + c;

const tax =
  (rate) =>
  (price) =>
    price * (1 + rate / 100);

const withGST = tax(18);
const withVAT = tax(10);

console.log(withGST(1000));
console.log(withVAT(1000));

Topic 6 — Partial Application


function greet(greeting, time, name) {
  return `${greeting} ${time}, ${name}!`;
}

const sayGoodMorning =
  greet.bind(null, "Good", "morning");

console.log(
  sayGoodMorning("Priya")
);

const partial =
  (fn, ...preset) =>
  (...rest) =>
    fn(...preset, ...rest);

const sayHelloEvening =
  partial(greet, "Hello", "evening");

console.log(
  sayHelloEvening("Riya")
);
